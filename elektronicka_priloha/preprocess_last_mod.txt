# function to convert nltk tag to wordnet tag
def nltk_tag_to_wordnet_tag(nltk_tag):
    # print(nltk_tag)
    if nltk_tag.startswith('J'):
        return wordnet.ADJ
    elif nltk_tag.startswith('V'):
        return wordnet.VERB
    elif nltk_tag.startswith('N'):
        return wordnet.NOUN
    elif nltk_tag.startswith('R'):
        return wordnet.ADV
    else:
        return None


def pos_tag_for_lemma(tag):
    if tag.startswith('V'):
        return 'v'
    if tag.startswith('N'):
        return 'n'
    if tag.startswith('J'):
        return 'a'
    if tag.startswith('R'):
        return 'r'
    else:
        return None

def getCleanTokens_custom(message, dostopwords=True, wordlength=2):
    # print(message)
        
    message = re.sub('[+\"\'-+]', ' ', message)
    message = re.sub('[.$]', '', message)
    message = re.sub('[*]', ' ', message)



    # # Handeling Unicodes
    # message = re.sub(r'\\u\d{0,}\w', ' ', message)

    # # Handeling Hyphen
    # message = re.sub(r'-{2,}', ' ', message)

    # # Handeling Equal To
    # message = re.sub(r'={2,}', ' ', message)

    #Making Changes in Preprocessing
    message=make_preprocess_changes(message)

    #print("Filtered Message - {}".format(message))
    #     #  remove underscores
    #     message=re.sub('^_{2,}_$','',message)

    # take care of comma separated words(phrases)
    m_phrases = message.split(',')
    m_tokens = []
    for m in m_phrases:
        m_tokens = m_tokens + m.split(' ')
    m_tokens = [w for w in m_tokens if not re.match(sp_format, w) and not re.match(num_format, w) and w != '']
    m_tokens = nltk.pos_tag(m_tokens)
    m_tokens = [wordnet_lemmatizer.lemmatize(w[0].lower(), pos_tag_for_lemma(w[1])).lower() for w in m_tokens if
                nltk_tag_to_wordnet_tag(w[1]) != None]

    singular_tokens = []
    for n in m_tokens:
        n = removespecialcharacteratend(n)
        if len(n) > 0 and n.isalnum() == True:
            inflectedWord = inflect.plural(n)
            if len(inflectedWord) < len(n):
                n = inflectedWord

        if len(n) <= wordlength or n in stop:
            continue

        singular_tokens.append(n)

    msg_tokens = singular_tokens

    return msg_tokens


def add_label(noise_sen):
    output = []
    for i, s in enumerate(noise_sen):
        output.append(TaggedDocument(s, ["sent_" + str(i)]))
    return output

def process_data(df, columnname):
    sent_list = df[columnname].apply(lambda x: getCleanTokens_custom(str(x).encode('ascii', 'ignore').decode())).tolist()

    return sent_list
