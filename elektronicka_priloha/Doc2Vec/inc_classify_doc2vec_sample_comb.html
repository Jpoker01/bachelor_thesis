def add_label(noise_sen):
    output = []
    for i, s in enumerate(noise_sen):
        output.append(TaggedDocument(s, ["noise_" + str(i)]))
    return output

print('Initialize the model parameters')
model_d2v = gensim.models.Doc2Vec(dm = dm, #dm = 1 for PV_DM,dm-0 for PV-DBOW
                        #dm_concat=, # for concatenation instead of mean or sum
                        vector_size = vector_size, #trying(100,200) no. of desired features
                        window = window, #trying(3,4,5) width of the context window                                  
                        min_count = min_count, #trying(2,3,4) Ignores all words with total frequency lower than 2.                                  
                        workers = 4, # no. of cores
                        negative = negative,
                        min_alpha = min_alpha,
                        alpha = alpha, # learning rate                                  
                        seed = seed,# for reproducibility
                        epochs = max_epochs
                        )
